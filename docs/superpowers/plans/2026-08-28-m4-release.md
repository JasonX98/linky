```dart
// lib/engine/engine_update.dart（实现要点）
class EngineVersion implements Comparable<EngineVersion> {
  EngineVersion(List<int> raw) : parts = List.unmodifiable(raw);
  final List<int> parts;

  static EngineVersion? tryParse(String s) {
    final parts = s.trim().split('.').map((x) => int.tryParse(x.trim())).toList();
    if (parts.isEmpty || parts.any((p) => p == null)) return null;
    return EngineVersion(parts.cast<int>());
  }

  bool isNewerThan(EngineVersion other) {
    final max = parts.length > other.parts.length ? parts.length : other.parts.length;
    for (var i = 0; i < max; i++) {
      final a = i < parts.length ? parts[i] : 0;
      final b = i < other.parts.length ? other.parts[i] : 0;
      if (a != b) return a > b;
    }
    return false;
  }

  @override
  int compareTo(EngineVersion other) => isNewerThan(other) ? 1 : (other.isNewerThan(this) ? -1 : 0);

  @override
  String toString() => parts.join('.');
}
```
